:error

=========================
RPCS: one method
=========================

rpc_service ServiceType {
  Read(RequestTable):ResponseTable;
}

----

(schema
    (rpc_service name: (ident)
        (rpc_method name: (ident) request: (ident) response: (ident))
    )
)

=========================
RPCS: two methods
=========================

rpc_service ServiceType {
  Read(ReadRequestTable):ReadResponseTable;
  Create(CreateRequestTable):CreateResponseTable;
}

----

(schema
    (rpc_service name: (ident)
        (rpc_method name: (ident) request: (ident) response: (ident))
        (rpc_method name: (ident) request: (ident) response: (ident))
    )
)

=========================
RPCS: attributes
=========================

rpc_service ServiceType (custom_attribute) {
  Read(ReadRequestTable):ReadResponseTable (custom_attribute);
}

----

(schema
    (rpc_service name: (ident) (metadata (attribute name: (ident)))
        (rpc_method name: (ident) request: (ident) response: (ident) (metadata (attribute name: (ident))))
    )
)

table HelloRequest {}
table HelloResponse {}

rpc_service HelloService {
  Hello(HelloRequest):HelloResponse;
  StreamClient(HelloRequest):HelloResponse (streaming: "client");
  StreamServer(HelloRequest):HelloResponse (streaming: "server");
  Stream(HelloRequest):HelloResponse (streaming: "bidi");
}
